-- ── Allgemeine Hilfsfunktionen ───────────────────────────────────────────
local function new(className, props, parent)
    local object = Instance.new(className)
    for property, value in pairs(props or {}) do
        object[property] = value
    end
    if parent then object.Parent = parent end
    return object
end

local function corner(object, radius)
    local item = Instance.new("UICorner")
    item.CornerRadius = UDim.new(0, radius or 8)
    item.Parent = object
    return item
end

local function stroke(object, color, thickness, transparency)
    local item = Instance.new("UIStroke")
    item.Color = color or Theme.Stroke
    item.Thickness = thickness or 1
    item.Transparency = transparency or 0
    item.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    item.Parent = object
    return item
end

local function tween(object, properties, duration)
    local item = TweenService:Create(
        object,
        TweenInfo.new(duration or 0.14, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
        properties
    )
    item:Play()
    return item
end

local function fontSize(size)
    return math.floor((size or 11) * FONT_SCALE + 0.5)
end

local function makeLabel(parent, text, size, color, font, alignment)
    return new("TextLabel", {
        BackgroundTransparency = 1,
        Text = text,
        TextSize = fontSize(size),
        TextColor3 = color or Theme.Text,
        Font = font or Enum.Font.GothamMedium,
        TextXAlignment = alignment or Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
    }, parent)
end

-- Activated deckt Maus und Touch ab. Mehrfachbindungen werden vermieden.
local function bindClick(button, callback)
    button.Activated:Connect(callback)
end

local function applyGlass(frame, baseColor, transparency)
    local amount = transparency or 0.3
    frame.BackgroundColor3 = baseColor
    frame.BackgroundTransparency = amount

    local gradient = Instance.new("UIGradient")
    gradient.Rotation = 90
    gradient.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.05),
        NumberSequenceKeypoint.new(0.5, amount),
        NumberSequenceKeypoint.new(1, math.min(amount + 0.15, 0.9)),
    })
    gradient.Parent = frame

    local rim = new("Frame", {
        Size = UDim2.new(1, -16, 0, 1),
        Position = UDim2.new(0, 8, 0, 1),
        BackgroundColor3 = Color3.fromRGB(255, 255, 255),
        BackgroundTransparency = 0.65,
        BorderSizePixel = 0,
        ZIndex = frame.ZIndex + 1,
    }, frame)

    local rimGradient = Instance.new("UIGradient")
    rimGradient.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 1),
        NumberSequenceKeypoint.new(0.5, 0.4),
        NumberSequenceKeypoint.new(1, 1),
    })
    rimGradient.Parent = rim
end

local function escapeRichText(value)
    local text = tostring(value or "")
    text = text:gsub("&", "&amp;")
    text = text:gsub("<", "&lt;")
    text = text:gsub(">", "&gt;")
    text = text:gsub('"', "&quot;")
    text = text:gsub("'", "&apos;")
    return text
end

local connections = {}
local viewportConnection

local function track(connection)
    table.insert(connections, connection)
    return connection
end

local function untrack(connection)
    if not connection then return end
    for index = #connections, 1, -1 do
        if connections[index] == connection then
            table.remove(connections, index)
            break
        end
    end
    pcall(function() connection:Disconnect() end)
end

local function recalculateScale()
    local camera = workspace.CurrentCamera
    if not camera then return end

    local height = camera.ViewportSize.Y
    local scale
    if UserInputService.TouchEnabled then
        scale = math.clamp(height / 1000, 0.62, 0.88)
    else
        scale = math.clamp(height / 800, 0.70, 1.05)
    end

    GlobalScale.Scale = scale
    Master.Size = UDim2.new(1 / scale, 0, 1 / scale, 0)
end

local function setupCamera()
    if viewportConnection then
        viewportConnection:Disconnect()
        viewportConnection = nil
    end

    local camera = workspace.CurrentCamera
    if camera then
        viewportConnection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(recalculateScale)
    end
    recalculateScale()
end

track(workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(setupCamera))
task.spawn(setupCamera)

    end)

local function logLine(message, color)
    local safeMessage = escapeRichText(message)
    local chosenColor = color or Theme.Text
    local colorTag = string.format(
        "rgb(%d,%d,%d)",
        math.floor(chosenColor.R * 255),
        math.floor(chosenColor.G * 255),
        math.floor(chosenColor.B * 255)
    )

    table.insert(logLines, '<font color="' .. colorTag .. '">' .. safeMessage .. "</font>")
    if #logLines > MAX_LOG then
        table.remove(logLines, 1)
    end

    if ConsoleText then
        ConsoleText.Text = table.concat(logLines, "\n")
        scrollConsoleToBottom()
    end
end

-- ── P-Schaltfläche bewegen: Touch und Maus ────────────────────────────────
local mobileButtonWasDragged = false
do
    local moving = false
    local moveInput
    local startPoint
    local startAbsolute

    track(MobileToggle.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.Touch
            and input.UserInputType ~= Enum.UserInputType.MouseButton1 then
            return
        end
        moving = true
        moveInput = input
        mobileButtonWasDragged = false
        startPoint = Vector2.new(input.Position.X, input.Position.Y)
        startAbsolute = MobileToggle.AbsolutePosition
    end))

    track(UserInputService.InputChanged:Connect(function(input)
        if not moving or not moveInput then return end
        local isTouch = moveInput.UserInputType == Enum.UserInputType.Touch and input == moveInput
        local isMouse = moveInput.UserInputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseMovement
        if not isTouch and not isMouse then return end

        local delta = Vector2.new(input.Position.X, input.Position.Y) - startPoint
        if delta.Magnitude > 5 then mobileButtonWasDragged = true end
        local scale = GlobalScale.Scale
        if scale <= 0 then scale = 1 end
        local camera = workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or Vector2.new(800, 600)
        local size = MobileToggle.AbsoluteSize
        local absoluteX = math.clamp(
            startAbsolute.X + delta.X,
            8,
            math.max(8, viewport.X - size.X - 8)
        )
        local absoluteY = math.clamp(
            startAbsolute.Y + delta.Y,
            8,
            math.max(8, viewport.Y - size.Y - 8)
        )
        MobileToggle.Position = UDim2.new(
            0, absoluteX / scale + size.X / scale,
            0, absoluteY / scale + size.Y / scale
        )
    end))

    track(UserInputService.InputEnded:Connect(function(input)
        if moveInput and input == moveInput then
            moving = false
            moveInput = nil
        end
    end))
end

-- ── Fenster ziehen: Maus und Touch ───────────────────────────────────────
do
    local dragging = false
    local dragInput
    local dragStart
    local dragStartAbsolute
    local startPosition
    local moved = false
    local threshold = UserInputService.TouchEnabled and 8 or 2

    local function overControl(position)
        for _, control in ipairs({ MinBtn, CloseBtn }) do
            local p = control.AbsolutePosition
            local s = control.AbsoluteSize
            if position.X >= p.X - 6 and position.X <= p.X + s.X + 6
                and position.Y >= p.Y - 6 and position.Y <= p.Y + s.Y + 6 then
                return true
            end
        end
        return false
    end

    track(Header.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then
            return
        end
        if dragging or overControl(input.Position) then return end

        dragging = true
        dragInput = input
        dragStart = Vector2.new(input.Position.X, input.Position.Y)
        dragStartAbsolute = Window.AbsolutePosition
        startPosition = Window.Position
        moved = false

        local endConnection
        endConnection = track(input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End
                or input.UserInputState == Enum.UserInputState.Cancel then
                if dragInput == input then
                    dragging = false
                    dragInput = nil
                end
                if endConnection then
                    untrack(endConnection)
                    endConnection = nil
                end
            end
        end))
    end))

    track(UserInputService.InputChanged:Connect(function(input)
        if not dragging or not dragInput then return end

        local isTouchMove = dragInput.UserInputType == Enum.UserInputType.Touch and input == dragInput
        local isMouseMove = dragInput.UserInputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseMovement
        if not isTouchMove and not isMouseMove then return end

        local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStart
        if not moved then
            if delta.Magnitude < threshold then return end
            moved = true
        end

        local scale = GlobalScale.Scale
        if scale <= 0 then scale = 1 end

        local camera = workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or Vector2.new(800, 600)
        local absoluteSize = Window.AbsoluteSize
        local clampedX = math.clamp(
            dragStartAbsolute.X + delta.X,
            8,
            math.max(8, viewport.X - absoluteSize.X - 8)
        )
        local clampedY = math.clamp(
            dragStartAbsolute.Y + delta.Y,
            8,
            math.max(8, viewport.Y - absoluteSize.Y - 8)
        )

        Window.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + (clampedX - dragStartAbsolute.X) / scale,
            startPosition.Y.Scale,
            startPosition.Y.Offset + (clampedY - dragStartAbsolute.Y) / scale
        )
    end))
end

    minimised = state

    end
    applyTitleLayout(minimised)

    local targetWidth = minimised and 270 or WIN_W
    windowTween = TweenService:Create(

        { Size = UDim2.fromOffset(targetWidth, minimised and MIN_H or (compactMode and 460 or WIN_H)) }
    )

    contentTween = TweenService:Create(
        ContentGroup,
        TweenInfo.new(0.20, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
        { GroupTransparency = minimised and 1 or 0 }
    )

    windowTween:Play()
    contentTween:Play()
    MinBtn.Text = minimised and "+" or "–"
end

    if not Window.Visible then

        setMinimised(false)
    else
        Window.Visible = false
    end
end)

-- ── Cleanup / Close ──────────────────────────────────────────────────────
local stopped = false

local function stopEverything()
    if stopped then return end
    stopped = true

    if viewportConnection then
        pcall(function() viewportConnection:Disconnect() end)
        viewportConnection = nil
    end
    if glowTween then
        pcall(function() glowTween:Cancel() end)
        glowTween = nil
    end

    for _, connection in ipairs(connections) do
        pcall(function() connection:Disconnect() end)
    end
    table.clear(connections)

    if GUI and GUI.Parent then

local function runWithTimeout(seconds, work)
    local finished = false
    task.spawn(function()
        pcall(work)
        finished = true
    end)
    local startedAt = os.clock()
    while not finished and os.clock() - startedAt < seconds do
        task.wait(0.1)
    end
    return finished
end

local function loadBackgroundImage()
    if not GalaxyImage then return end

    local assetNumber = tostring(GALAXY_ASSET_ID):match("%d+")
    if not assetNumber then
        logLine("Background: no valid asset ID set.", Theme.Warn)
        return
    end
    logLine("Background: loading " .. GALAXY_ASSET_ID, Theme.Dim)

    local candidates = {}

    -- Ist die ID ein Decal, liefert das Decal die echte Bild-URL (falls GetObjects verfügbar ist).
    local objects
    runWithTimeout(8, function()
        objects = game:GetObjects("rbxassetid://" .. assetNumber)
    end)
    if type(objects) == "table" then
        for _, object in ipairs(objects) do
            local okTexture, texture = pcall(function()
                if object:IsA("Decal") or object:IsA("Texture") then
                    return object.Texture
                end
                return nil
            end)
            if okTexture and type(texture) == "string" and texture ~= "" then
                table.insert(candidates, { texture, "decal image" })
            end
            pcall(function() object:Destroy() end)
        end
    end
    table.insert(candidates, { GALAXY_ASSET_ID, "rbxassetid" })
    table.insert(candidates, { "http://www.roblox.com/asset/?id=" .. assetNumber, "asset url" })

    for _, candidate in ipairs(candidates) do
        if not GalaxyImage.Parent then return end
        local image, label = candidate[1], candidate[2]

        GalaxyImage.Image = image
        local fetchOk
        local finished = runWithTimeout(10, function()
            ContentProvider:PreloadAsync({ GalaxyImage }, function(_, status)
                fetchOk = (status == Enum.AssetFetchStatus.Success)
            end)
        end)

        if fetchOk == true or GalaxyImage.IsLoaded then
            logLine("Background loaded (" .. label .. ").", Theme.Emerald)
            return
        end

        local reason = "timeout"
        if finished then
            reason = (fetchOk == false) and "fetch failed" or "no result"
        end
        logLine("Background attempt '" .. label .. "' failed: " .. reason .. ".", Theme.Warn)
    end

    GalaxyImage.Image = ""
    logLine("Background not loadable — procedural nebula stays.", Theme.Red)
    logLine("Check asset type, moderation status and access in Creator Hub.", Theme.Dim)
end

local B64_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

local function decodeBase64(data)
    local lookup = {}
    for i = 1, #B64_CHARS do
        lookup[string.byte(B64_CHARS, i)] = i - 1
    end
    data = string.gsub(data, "[^%w%+/=]", "")
    local chunks, count = {}, 0
    for i = 1, #data, 4 do
        local a, b, c, d = string.byte(data, i, i + 3)
        local va, vb = lookup[a], lookup[b]
        if not va or not vb then break end
        local vc = c and lookup[c]
        local vd = d and lookup[d]
        local triple = va * 262144 + vb * 4096 + (vc or 0) * 64 + (vd or 0)
        local b1 = math.floor(triple / 65536)
        local b2 = math.floor(triple / 256) % 256
        local b3 = triple % 256
        count = count + 1
        if vc == nil then
            chunks[count] = string.char(b1)
        elseif vd == nil then
            chunks[count] = string.char(b1, b2)
        else
            chunks[count] = string.char(b1, b2, b3)
        end
    end
    return table.concat(chunks)
end

local function loadTitleImage()
    -- Setzt das Bild, lädt es vor und meldet, ob Roblox es liefern konnte.
    local function confirmLoaded(image)
        TitleImage.Image = image
        local fetchOk
        local finished = runWithTimeout(10, function()
            ContentProvider:PreloadAsync({ TitleImage }, function(_, status)
                fetchOk = (status == Enum.AssetFetchStatus.Success)
            end)
        end)
        if fetchOk == true or TitleImage.IsLoaded then
            return true
        end
        local startedAt = os.clock()
        while not TitleImage.IsLoaded and os.clock() - startedAt < 2 do
            task.wait(0.1)
        end
        if TitleImage.IsLoaded then
            return true
        end
        if finished then
            return false, (fetchOk == false) and "fetch failed" or "no result"
        end
        return false, "timeout"
    end

    local function activate(label)
        if not TitleImage.Parent then return end
        titleActive = true
        TitleImage.ImageTransparency = 0
        Brand.Visible = false
        Subtitle.Visible = false
        applyTitleLayout(minimised)
        logLine("Title image loaded (" .. label .. ").", Theme.Emerald)
    end

    -- 1) Roblox-Asset-ID (nur wenn TITLE_ASSET_ID gesetzt ist)
    if TITLE_ASSET_ID ~= "" then
        local ok, reason = confirmLoaded(TITLE_ASSET_ID)
        if ok then
            activate("asset id")
            return
        end
        logLine("Title asset ID failed: " .. tostring(reason) .. ".", Theme.Warn)
    end

    -- 2) Eingebettetes PNG über writefile + getcustomasset (Executor-Funktionen)
    if type(writefile) ~= "function" or type(getcustomasset) ~= "function" then
        logLine("Title image: writefile/getcustomasset not available — text title stays.", Theme.Warn)
        return
    end
    local raw = decodeBase64(TITLE_PNG_B64)
    if #raw < 1000 or string.sub(raw, 2, 4) ~= "PNG" then
        logLine("Title image: embedded data could not be decoded.", Theme.Red)
        return
    end
    local fileName = "phantix_title_v034.png"
    local okWrite, errWrite = pcall(writefile, fileName, raw)
    if not okWrite then
        logLine("Title image: writefile failed (" .. tostring(errWrite) .. ").", Theme.Red)
        return
    end
    local okAsset, asset = pcall(getcustomasset, fileName)
    if not okAsset or type(asset) ~= "string" or asset == "" then
        logLine("Title image: getcustomasset failed (" .. tostring(asset) .. ").", Theme.Red)
        return
    end
    local ok, reason = confirmLoaded(asset)
    if ok then
        activate("embedded")
        return
    end
    TitleImage.Image = ""
    logLine("Title image not confirmed (" .. tostring(reason) .. ") — text title stays.", Theme.Warn)
end
